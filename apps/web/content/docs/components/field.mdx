## Usage

```tsx
import {

  Field,
  FieldDescription,
  FieldLabel,
} from "@/components/ui/field"

export function Example() {
  return (
    <Field>
      <FieldLabel htmlFor="email">Email</FieldLabel>
      <Input id="email" type="email" />
      <FieldDescription>We'll never share your email.</FieldDescription>
    </Field>
  )
}
```
