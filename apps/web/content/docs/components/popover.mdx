## Usage

```tsx
import {

  Popover,
  PopoverContent,
  PopoverTrigger,
} from "@/components/ui/popover"

export function Example() {
  return (
    <Popover>
      <PopoverTrigger render={<Button variant="outline">Open</Button>} />
      <PopoverContent>Place content for the popover here.</PopoverContent>
    </Popover>
  )
}
```
