## Usage

```tsx
import { AspectRatio } from "@/components/ui/aspect-ratio"

export function Example() {
  return (
    <div className="w-64">
      <AspectRatio ratio={16 / 9}>
        <img
          src="/image.jpg"
          alt="Photo"
          className="size-full rounded-lg object-cover"
        />
      </AspectRatio>
    </div>
  )
}
```
