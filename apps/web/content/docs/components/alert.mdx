## Usage

```tsx
import {

  Alert,
  AlertDescription,
  AlertTitle,
} from "@/components/ui/alert"

export function Example() {
  return (
    <Alert>
      <Icon />
      <AlertTitle>Heads up!</AlertTitle>
      <AlertDescription>
        You can add components to your app.
      </AlertDescription>
    </Alert>
  )
}
```
